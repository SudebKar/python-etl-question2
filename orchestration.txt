Leader election design (simple & safe)
Core idea

One row = one active leader

Leader must renew heartbeat

If heartbeat expires → another pod can take over

Step 1️⃣ Create leader lock table
CREATE TABLE scheduler_leader_lock (
    lock_name        VARCHAR(50) PRIMARY KEY,
    owner_id         VARCHAR(100),
    last_heartbeat   DATETIME
);


You will only ever have one row:

lock_name = 'SCHEDULER_LEADER'

Step 2️⃣ LeaderElectionService (core logic)

📁 src/core/leader_election.py

import socket
from datetime import datetime, timedelta
from sqlmodel import text
from src.db.session import get_session


class LeaderElectionService:

    LOCK_NAME = "SCHEDULER_LEADER"
    HEARTBEAT_TIMEOUT_SEC = 60

    @staticmethod
    def _owner_id():
        return socket.gethostname()

    @classmethod
    def try_acquire_or_renew(cls) -> bool:
        """
        Returns True if this pod is leader.
        """

        now = datetime.utcnow()
        expiry = now - timedelta(seconds=cls.HEARTBEAT_TIMEOUT_SEC)

        with get_session() as session:
            session.exec(text("""
                MERGE scheduler_leader_lock AS target
                USING (SELECT :lock_name AS lock_name) AS source
                ON target.lock_name = source.lock_name
                WHEN MATCHED AND (
                    target.last_heartbeat < :expiry
                    OR target.owner_id = :owner_id
                )
                THEN UPDATE SET
                    owner_id = :owner_id,
                    last_heartbeat = :now
                WHEN NOT MATCHED
                THEN INSERT (lock_name, owner_id, last_heartbeat)
                VALUES (:lock_name, :owner_id, :now);
            """), {
                "lock_name": cls.LOCK_NAME,
                "owner_id": cls._owner_id(),
                "now": now,
                "expiry": expiry
            })

            session.commit()

            result = session.exec(text("""
                SELECT owner_id
                FROM scheduler_leader_lock
                WHERE lock_name = :lock_name
            """), {"lock_name": cls.LOCK_NAME}).first()

            return result and result[0] == cls._owner_id()


This:

is atomic

is race-safe

works across regions

survives crashes

Step 3️⃣ Modify scheduler bootstrap to respect leadership

📁 src/core/scheduler.py

from apscheduler.schedulers.background import BackgroundScheduler
from src.core.job_registry import JobRegistry
from src.jobs.status_monitor_job import StatusMonitorJob
from src.core.leader_election import LeaderElectionService
import time


def start_scheduler():
    scheduler = BackgroundScheduler()
    scheduler.start()

    while True:
        is_leader = LeaderElectionService.try_acquire_or_renew()

        if is_leader and not scheduler.running:
            scheduler.start()
            JobRegistry.register_jobs(scheduler)
            scheduler.add_job(
                StatusMonitorJob.execute,
                trigger="interval",
                minutes=1,
                id="status_monitor",
                replace_existing=True
            )

        if not is_leader and scheduler.running:
            scheduler.shutdown(wait=False)

        time.sleep(30)


👉 Only one pod will ever schedule jobs.
