from sqlmodel import select
from src.db.session import get_session
from src.models.scheduler_process_schedule import SchedulerProcessSchedule

class ScheduleRepository:

    @staticmethod
    def get_active_schedules():
        with get_session() as session:
            stmt = select(SchedulerProcessSchedule).where(
                SchedulerProcessSchedule.active_flag == True
            )
            return session.exec(stmt).all()

4.2 Run state repository
# src/repositories/run_state_repository.py
from sqlmodel import select
from datetime import datetime
from src.db.session import get_session
from src.models.scheduler_run_state import SchedulerRunState

class RunStateRepository:

    @staticmethod
    def create_run(run: SchedulerRunState):
        with get_session() as session:
            session.add(run)
            session.commit()

    @staticmethod
    def update_status(run_id: str, status: str, error: str = None):
        with get_session() as session:
            run = session.get(SchedulerRunState, run_id)
            if not run:
                return
            run.status = status
            if status in ("COMPLETED", "FAILED"):
                run.end_time = datetime.utcnow()
            if error:
                run.error_description = error
            session.add(run)
            session.commit()

from sqlmodel import select
from src.db.session import get_session
from src.models.scheduler_process_dependency import SchedulerProcessDependency

class DependencyRepository:

    def get_dependencies(process_id: str):
